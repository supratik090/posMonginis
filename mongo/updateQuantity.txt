db.inventories.updateMany(
  {
    code: {
      $in: db.items.aggregate([
        {
          $lookup: {
            from: "inventories",
            localField: "code",
            foreignField: "code",
            as: "inventoryDetails"
          }
        },
        { $unwind: "$inventoryDetails" },
        {
          $addFields: {
            shelfLifeDate: {
              $toDate: {
                $add: [
                  { $toLong: "$inventoryDetails.createdAt" },
                  { $multiply: ["$shelfLife", 86400000] }
                ]
              }
            },
            todaysDate: {
              $dateTrunc: {
                date: "$$NOW",
                unit: "day"
              }
            },
            shelfLifeDayOnly: {
              $dateTrunc: {
                date: "$shelfLifeDate",
                unit: "day"
              }
            }
          }
        },
        {
          $match: {
            $expr: {
              $lt: ["$shelfLifeDayOnly", "$todaysDate"]  // Expired only (excluding today)
            }
          }
        },
        { $project: { code: 1, _id: 0 } }
      ]).toArray().map(doc => doc.code)
    },
    quantity: { $gt: 0 }  // Only update if quantity > 0
  },
  {
    $set: { quantity: 0 }
  }
);
